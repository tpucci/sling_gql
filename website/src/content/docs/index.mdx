export const schemaSnippet = `type Launch {
  id: ID!
  name: String!
  details: String
  status: LaunchStatus!
  rocket: Rocket!
  payloads: [Payload!]!

export const generatedSnippet = `class Launch extends Accessor {
  String? get id => scalar<String>('id');
  String? get name => scalar<String>('name');
  set name(String? v) => write('name', v);
  String? get details => scalar<String>('details');
  String? get status => scalar<String>('status');
  Rocket? get rocket => object('rocket', Rocket.new);
  List<Payload>? get payloads => list('payloads', Payload.new);
}`;

export const gqlSnippet = `query {
  latestLaunch {
    __typename
    name
    rocket {
      __typename
      name
    }
  }
}`;

<section class="landing-section">
  <h2>How it works</h2>
  <p class="lead">Four moving parts, ~600 lines of runtime, one dependency (<code>http</code>).</p>
  <ol class="steps">
    <li>
      <strong>Generated accessors, not proxies</strong>
      <p>One Dart class per GraphQL type. Every getter records the field on a selection tree, then reads the cache. Static, typed, tree-shakeable.</p>
    </li>
    <li>
      <strong>Skeletons instead of loading branches</strong>
      <p>Before data arrives, scalars are <code>null</code> and lists have one placeholder item. Your layout renders once, with the same code path.</p>
    </li>
    <li>
      <strong>One request per frame</strong>
      <p>Selections from every widget — including lazily built list rows — merge into a single document flushed at the end of the frame.</p>
    </li>
    <li>
      <strong>Targeted rebuilds</strong>
      <p>The response is deep-merged into a path-addressed cache; only widgets that read the affected fields are asked to rebuild.</p>
    </li>
  </ol>
</section>

<section class="landing-section">
  <h2>A typed API, generated from your schema</h2>
  <p class="lead">
    One Dart class per GraphQL type. Every field is a real member — autocomplete, jump to
    definition, doc comments from the schema. Every field is nullable, because it may not be
    fetched yet: the compiler makes you handle the skeleton state instead of discovering it at runtime.
  </p>
